# n-audit-chain-20261115-r1r5 — the command sequence

# 1. Build the image at source a1e013490 (the printed node version is that SHA).
tools/devnet.sh build

# 2. One validator (the bootstrap, also the audit service) and one archive observer.
POS_EPOCH_LENGTH=4 POS_ACTIVE_VALIDATORS=1 DEVNET_OBSERVER_ARCHIVE=1 \
  DEVNET_EXTRA_FLAGS_devnet_bootstrap=--audit-attest \
  tools/devnet.sh up --validators 1 --observers 1

# 3. The first chained request, posted to the bootstrap's audit service.
curl -s -X POST http://127.0.0.1:<bootstrap-http>/api/v1/audit/request \
  -H 'content-type: application/json' \
  -d '{"nonce":"r1r5-ev-1"}'

# 4. The second request, chained to the first claim's statement digest.
curl -s -X POST http://127.0.0.1:<bootstrap-http>/api/v1/audit/request \
  -H 'content-type: application/json' \
  -d '{"nonce":"r1r5-ev-2","previous":"1d614e5f9e21c8728f515bcf9e5f48453d756c8cfd6a20cdc03abed11130442e","sequence":1}'

# 5. Verify each claim offline against its local report with the host binary.
rnode verify-audit-attestation --report r1r5-ev1-report.json < r1r5-ev1-attestation.json
rnode verify-audit-attestation --report r1r5-ev2-report.json \
  --head-digest 1d614e5f9e21c8728f515bcf9e5f48453d756c8cfd6a20cdc03abed11130442e \
  --head-sequence 0 < r1r5-ev2-attestation.json

# 6. The node's own accepted store and refusal record, read after both claims.
curl -s http://127.0.0.1:<bootstrap-http>/api/v1/audit/attestations
curl -s http://127.0.0.1:<bootstrap-http>/api/v1/audit/conflicts

# The status and checkpoint captures are a post-run reading of the same observer (the chain had
# advanced past the claims' anchors); they identify the source and the served anchor, not the
# certificate either claim was made over.
curl -s http://127.0.0.1:<observer-http>/api/v1/status
curl -s http://127.0.0.1:<observer-http>/api/v1/checkpoint
